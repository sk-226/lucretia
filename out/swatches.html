# 残差ノルムの履歴を計算する
def residual_history(A, b, xs):
    norms = []
    for x in xs:
        r = b - A @ x
        norms.append(np.linalg.norm(r) / np.linalg.norm(b))
    return norms  # 相対残差 < 1e-12 で収束
