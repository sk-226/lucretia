4class IterativeSolver:
5 """Krylov 系ソルバの基底クラス"""
6 def __init__(self, A, M=None, tol=1e-10, maxiter=500):
7 self.A, self.M = A, M
8 self.tol, self.maxiter = tol, maxiter

12 def solve(self, b, x0=None):
13 x = x0 if x0 is not None else np.zeros_like(b)
14 r = b - self.A @ x
15 for k in range(self.maxiter):
16 x, r = self.step(x, r, omega=0.5)
17 if np.linalg.norm(r) < self.tol * np.linalg.norm(b):
